- [x] **Step 2.7:现代化 `calculateMD5`**

把:
```js
export function calculateMD5(file) {
  return new Promise((resolve, reject) => {
    const chunkSize = 2097152; // 2MB
    const chunk = file.slice(0, chunkSize);
    const fileReader = new FileReader();
    fileReader.onload = e => resolve(SparkMD5.ArrayBuffer.hash(e.target.result));
    fileReader.onerror = reject;
    fileReader.readAsArrayBuffer(chunk);
  });
}
```
改为:
```js
// 只 hash 前 2MB(性能换碰撞率)。chunkSize 必须保持 2097152,不可改全文件分片(老 IDB key 兼容)。
export async function calculateMD5(file) {
  const chunkSize = 2097152; // 2MB
  const buffer = await file.slice(0, chunkSize).arrayBuffer();
  return SparkMD5.ArrayBuffer.hash(buffer);
}
```
(保留 `chunkSize = 2097152` 语义;`SparkMD5` import 不变。)
